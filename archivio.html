function openGeniusScan() { const isAndroid = /Android/.test(navigator.userAgent); const isIOS = /iPhone|iPad|iPod/.test(navigator.userAgent); let timeoutId = null, hasLostFocus = false; const onBlur = () => { hasLostFocus = true; if (timeoutId) clearTimeout(timeoutId); window.removeEventListener('blur', onBlur); }; window.addEventListener('blur', onBlur); if (isAndroid) { window.location.href = "intent://scan/#Intent;package=com.thegrizzlylabs.geniusscan.free;scheme=zxing;end"; timeoutId = setTimeout(() => { if (!hasLostFocus && confirm("Genius Scan non si è aperto. Scaricare dal Play Store?")) window.open("https://play.google.com/store/apps/details?id=com.thegrizzlylabs.geniusscan.free", "_blank"); window.removeEventListener('blur', onBlur); }, 5000); } else if (isIOS) { window.location.href = "geniuscan://"; timeoutId = setTimeout(() => { if (!hasLostFocus && confirm("Genius Scan non si è aperto. Scaricare dall'App Store?")) window.open("https://apps.apple.com/app/genius-scan/id377672876", "_blank"); window.removeEventListener('blur', onBlur); }, 5000); } else { alert("Link solo su mobile."); } }
